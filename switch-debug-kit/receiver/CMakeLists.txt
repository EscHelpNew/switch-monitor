cmake_minimum_required(VERSION 3.16)
if(NOT EXISTS "${NXDK_CONFIG_DIR}/reload_project.h")
    message(FATAL_ERROR "Set NXDK_CONFIG_DIR to the directory generated by: python build.py --config-only")
endif()
add_library(switch_debug_receiver STATIC
    reload_core.c reload_bridge.c remote_reload.c reload_launch.c reload_reports.c reload_main.c)
target_include_directories(switch_debug_receiver PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}" "${NXDK_CONFIG_DIR}")
target_compile_features(switch_debug_receiver PUBLIC c_std_11)
